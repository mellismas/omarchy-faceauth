# pam_faceauth returns PAM_IGNORE for every failure (no daemon, timeout, no match,
# malformed reply, panic). That is what makes it safe in a stack that has a
# password behind it. A stack with NOTHING behind it is still closed with
# pam_deny, as belt and braces: on Linux-PAM 1.7.2 an auth stack whose only
# module returned PAM_IGNORE answers PAM_PERM_DENIED (measured against a stopped
# backend), so an ignore-only stack fails closed on its own, but that is a
# library behaviour and not a promise, and `doctor` checks for the pam_deny line.

# /etc/pam.d/omarchy-lock-face   (the lock screen's third stack, beside -password and -fingerprint;
#                                 the password is a separate service on its own lane, so this one
#                                 must be closed)
#%PAM-1.0
auth       sufficient                  pam_faceauth.so socket=/run/faceauth/sock timeout=8
auth       required                    pam_deny.so
account    include                     system-local-login

# /etc/pam.d/sudo and /etc/pam.d/polkit-1: one line at the top, opt-in. The
# password lines below it are the fallback, so no pam_deny here. `consent`
# makes elevation a deliberate act: the daemon opens a window naming the
# command and the requester, and the request is approved by two nods or by
# the password typed into that window. The window waits until it is answered
# (`timeout=` is ignored on a consent line). Sitting in front of the machine
# never elevates anything by itself.
auth      sufficient pam_faceauth.so socket=/run/faceauth/sock consent
